                          )}
                          {d.status === 'PENDING_REVIEW' && (
                            <button className="btn btn-ghost sm" onClick={() => api.approveDocument(Number(id), d.id).then(() => fetchDocsLifecycle())} title="قبول" style={{padding:'4px 8px',display:'flex',alignItems:'center',justifyContent:'center',color:'var(--success)'}}>
